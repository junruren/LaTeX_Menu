\noindent
\begin{recipe}[half=batch]{
  name = Winter Aperol Spritz, name-zh = 冬日蔓越莓气泡酒,
  glass = stemless, abv = 7,
  yield = {≈ 16 杯 × 140 mL · 共 2.25 L},
  method = 饮料桶自取, serve = {派对杯 · 加冰}, garnish = {橙片 + 迷迭香},
}
\begin{ingredients}
  \ing{750 mL}{1 瓶}{Aperol}{Aperol 开胃酒}
  \ing{750 mL}{1 瓶}{Prosecco}{普罗塞克起泡酒}
  \ing{250 mL}{}{Unsweetened cranberry juice}{无糖纯蔓越莓汁}
  \ing{500 mL}{}{Club soda}{无糖苏打水}
  \ing{2–3 个}{}{Orange, thin half-moons}{鲜橙，切薄半月片}
  \ing{16–20 枝}{}{Rosemary sprigs}{迷迭香小枝}
\end{ingredients}
\prep{把 Aperol 和蔓越莓汁混合冷藏。Prosecco 和苏打水也提前冰好，等客人到了再开瓶。}
\serve{把调好的酒底倒进饮料桶，缓缓加入 Prosecco 和苏打水，轻轻搅匀。冰块放在杯里，每杯接约 140 mL，再放一片橙子和一小枝迷迭香。}
\end{recipe}\hfill
\begin{recipe}[half=batch]{
  name = November Nightcap, name-zh = 十一月的晚安酒,
  glass = rocks, abv = 24–26,
  yield = {6 杯 × 75 mL · 共 450 mL},
  method = 预调装瓶冷藏, serve = {小杯 · 不加冰}, garnish = 橙皮挤油,
}
\begin{ingredients}
  \ing{120 mL}{4 oz}{Bourbon}{波本威士忌}
  \ing{60 mL}{2 oz}{Martell Blue Swift}{马爹利 Blue Swift}
  \ing{90 mL}{3 oz}{Campari}{金巴利}
  \ing{90 mL}{3 oz}{Sweet vermouth (Cocchi di Torino)}{甜味美思}
  \ing{90 mL}{3 oz}{Cold water}{冷水（整批稀释量）}
  \ing{1 段}{2–3 cm}{Cinnamon stick}{肉桂棒}
  \ing{6 片}{}{Fresh orange peel}{新鲜橙皮，每杯 1 片}
\end{ingredients}
\prep{把肉桂棒放进甜味美思里，浸泡 20 分钟后取出。加入波本、Blue Swift、Campari 和冷水，搅匀、装瓶、冷藏。90 mL 冷水已经算好了整批的稀释量。}
\serve{每杯直接倒入冰镇好的酒 75 mL。取一片橙皮，橙色那面朝向酒面，轻轻一挤，让橙油落在酒面上，再把橙皮放入杯中。}
\end{recipe}\par

\begin{recipe}[half=r2]{
  name = Mandarin Cosmo, name-zh = 柑橘大都会, glass = coupe, abv = 15,
  method = 摇, serve = {冰过的鸡尾酒杯 · 不加冰}, garnish = 橙皮挤油,

\begin{recipe}[half=r2]{
  name = French 75, name-zh = 法式 75, glass = flute, abv = 15,
  method = {摇 + 顶气泡酒}, serve = 香槟杯, garnish = 柠檬皮,
  source = {IBA 官方 French 75；原方用香槟},
}
\begin{ingredients}
  \ing{30 mL}{1 oz}{Tanqueray No. TEN gin}{添加利 No. TEN 金酒}
  \ing{15 mL}{½ oz}{Fresh lemon juice}{鲜柠檬汁}
  \ing{15 mL}{½ oz}{Simple syrup}{糖浆}
  \ing{60 mL}{2 oz}{Prosecco, to top}{普罗塞克起泡酒（最后倒）}
\end{ingredients}
\method{金酒、柠檬汁、糖浆加冰摇匀，滤入香槟杯，再缓缓倒入 Prosecco，轻轻搅一下。}
\tip{Prosecco 比香槟甜，如果偏甜，把糖浆减到 10 mL。}
\end{recipe}\rowgap
